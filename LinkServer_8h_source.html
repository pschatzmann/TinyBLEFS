97 static bool resolveRelative(const std::string& p, std::string& out) {
98 out.clear();
99 size_t i = 0;
100 while (i <= p.size()) {
101 size_t j = p.find('/', i);
102 if (j == std::string::npos) j = p.size();
103 std::string seg = p.substr(i, j - i);
104 if (seg == "..") return false;
105 if (!seg.empty() && seg != ".") {
106 if (!out.empty()) out += '/';
107 out += seg;
108 }
109 i = j + 1;
110 }
111 return true;
112 }

131 void handle(const Message& req) {
132 Message rsp;
133 rsp.op = req.op | kResponseFlag;
134 rsp.reqId = req.reqId;
135 rsp.status = (uint8_t)dispatch((Op)req.op, req.payload, rsp.payload);
136 // "not found", end of a listing and "is a directory" are normal answers
137 Status st = (Status)rsp.status;
138 if (st == Status::Ok || st == Status::NotFound || st == Status::EndOfList ||
139 st == Status::IsDirectory || st == Status::Busy)
140 TINYBLEFS_LOGD("server", "%s -> %s (%u bytes)", opName(req.op), statusName(st),
141 (unsigned)rsp.payload.size());
142 else
143 TINYBLEFS_LOGW("server", "%s failed: %s", opName(req.op), statusName(st));
144 if (rsp.status != (uint8_t)Status::Ok) rsp.payload.clear();
145 if (!sendMessage(t_, rsp))
146 TINYBLEFS_LOGE("server", "cannot send response to %s", opName(req.op));
147 }

155 Status dispatch(Op op, const std::vector<uint8_t>& in,
156 std::vector<uint8_t>& out) {
157 std::string path, path2;
158 FsEntry e;
159 switch (op) {
160 case Op::Info:
161 out.push_back(kProtocolVersion);
162 out.push_back((uint8_t)slots_.size());
163 out.push_back(read_only_ ? 1 : 0);
164 putStr(out, name_);
165 return Status::Ok;
166
167 case Op::Open: {
168 if (in.size() < 2) return badRequest("open: message too short");
169 if (!resolve(str(in, 1), path)) return badPath("open", str(in, 1));
170 uint8_t flags = in[0];
171 bool wr = flags & (kWrite | kAppend | kCreate | kTrunc);
172 if (wr && read_only_) return denied("open for writing", path);
173 bool exists = fs_.stat(path.c_str(), e);
174 if (exists && e.isDir) return Status::IsDirectory;
175 if (!exists && !(flags & (kCreate | kTrunc))) return Status::NotFound;
176 Slot* free_slot = nullptr;
177 uint8_t handle = 0;
178 for (size_t i = 0; i < slots_.size(); i++) {
179 if (!slots_[i].file) {
180 free_slot = &slots_[i];
181 handle = (uint8_t)i;
182 break;
183 }
184 }
185 if (!free_slot) {
186 TINYBLEFS_LOGW("server", "open %s: all %u file handles are in use",
187 path.c_str(), (unsigned)slots_.size());
188 return Status::NoHandle;
189 }
190 bool truncate = flags & kTrunc;
191 if (!locks_->acquire(path, wr, truncate)) return busy("open", path);
192 FsFile* f = fs_.open(path.c_str(), flags);
193 if (!f) {
194 locks_->release(path, wr);
195 TINYBLEFS_LOGE("server", "cannot open %s", path.c_str());
196 return Status::IoError;
197 }
198 free_slot->file = f;
199 free_slot->flags = flags;
200 free_slot->path = path;
201 free_slot->write = wr;
202 out.push_back(handle);
203 put32(out, f->size());
204 return Status::Ok;
205 }
206
207 case Op::Close: {
208 Slot* s = slot(in);
209 if (!s) return badHandle("close", in);
210 releaseSlot(*s);
211 return Status::Ok;
212 }
213
214 case Op::Read: {
215 Slot* s = slot(in);
216 if (!s) return badHandle("read", in);
217 if (in.size() < 7) return badRequest("read: message too short");
218 size_t len = get16(&in[5]);
219 out.resize(len);
220 int n = len ? s->file->read(get32(&in[1]), out.data(), len) : 0;
221 if (n < 0) {
222 TINYBLEFS_LOGE("server", "read error at offset %u", (unsigned)get32(&in[1]));
223 return Status::IoError;
224 }
225 out.resize(n);
226 return Status::Ok;
227 }
228
229 case Op::Write: {
230 Slot* s = slot(in);
231 if (!s) return badHandle("write", in);
232 if (in.size() < 5) return badRequest("write: message too short");
233 if (read_only_ || !(s->flags & (kWrite | kAppend | kCreate | kTrunc)))
234 return denied("write to a file opened read-only or on a read-only server", "");
235 size_t len = in.size() - 5;
236 int n = len ? s->file->write(get32(&in[1]), &in[5], len) : 0;
237 if (n < 0) {
238 TINYBLEFS_LOGE("server", "write error at offset %u", (unsigned)get32(&in[1]));
239 return Status::IoError;
240 }
241 if ((size_t)n < len)
242 TINYBLEFS_LOGW("server", "short write: %d of %u bytes (file system full?)", n,
243 (unsigned)len);
244 put16(out, (uint16_t)n);
245 return Status::Ok;
246 }
247
248 case Op::Stat:
249 if (!resolve(str(in, 0), path)) return badPath("stat", str(in, 0));
250 if (!fs_.stat(path.c_str(), e)) return Status::NotFound;
251 out.push_back(e.isDir ? 1 : 0);
252 put32(out, e.size);
253 return Status::Ok;
254
255 case Op::List:
256 if (in.size() < 2) return badRequest("list: message too short");
257 if (!resolve(str(in, 2), path)) return badPath("list", str(in, 2));
258 if (!fs_.list(path.c_str(), get16(&in[0]), e))
259 return Status::EndOfList;
260 out.push_back(e.isDir ? 1 : 0);
261 put32(out, e.size);
262 putStr(out, e.name);
263 return Status::Ok;
264
265 case Op::ListBatch: {
266 if (in.size() < 4) return badRequest("list: message too short");
267 if (!resolve(str(in, 4), path)) return badPath("list", str(in, 4));
268 uint32_t index = get16(&in[0]);
269 size_t budget = get16(&in[2]);
270 if (budget > max_list_bytes_) budget = max_list_bytes_;
271 out.push_back(0); // flags, set below
272 bool end = false;
273 while (true) {
274 if (!fs_.list(path.c_str(), index, e)) {
275 end = true;
276 break;
277 }
278 if (e.name.size() > 255) e.name.resize(255);
279 size_t need = 6 + e.name.size();
280 // always return at least one entry, else a small budget would never progress
281 if (out.size() > 1 && out.size() + need > budget) break;
282 out.push_back(e.isDir ? 1 : 0);
283 put32(out, e.size);
284 out.push_back((uint8_t)e.name.size());
285 putStr(out, e.name);
286 index++;
287 }
288 out[0] = end ? kListEnd : 0;
289 return Status::Ok;
290 }
291
292 case Op::Remove:
293 case Op::Mkdir:
294 case Op::Rmdir:
295 if (!resolve(str(in, 0), path)) return badPath(opName((uint8_t)op), str(in, 0));
296 if (read_only_) return denied(opName((uint8_t)op), path);
297 if (op == Op::Mkdir) {
298 if (fs_.stat(path.c_str(), e)) return Status::Exists;
299 if (fs_.mkdir(path.c_str())) return Status::Ok;
300 TINYBLEFS_LOGE("server", "mkdir %s failed", path.c_str());
301 return Status::IoError;
302 }
303 if (!fs_.stat(path.c_str(), e)) return Status::NotFound;
304 if (locks_->isBusy(path)) return busy(opName((uint8_t)op), path);
305 if (op == Op::Remove) {
306 if (e.isDir) {
307 TINYBLEFS_LOGW("server", "remove %s: is a directory (use rmdir)", path.c_str());
308 return Status::Error;
309 }
310 if (fs_.remove(path.c_str())) return Status::Ok;
311 TINYBLEFS_LOGE("server", "remove %s failed", path.c_str());
312 return Status::Error;
313 }
314 if (!e.isDir) {
315 TINYBLEFS_LOGW("server", "rmdir %s: not a directory", path.c_str());
316 return Status::Error;
317 }
318 if (fs_.rmdir(path.c_str())) return Status::Ok;
319 TINYBLEFS_LOGW("server", "rmdir %s failed (directory not empty?)", path.c_str());
320 return Status::Error;
321
322 case Op::Rename: {
323 if (in.size() < 2) return badRequest("rename: message too short");
324 size_t n = get16(&in[0]);
325 if (in.size() < 2 + n) return badRequest("rename: message too short");
326 std::string from((const char*)&in[2], n);
327 if (!resolve(from, path)) return badPath("rename", from);
328 if (!resolve(str(in, 2 + n), path2)) return badPath("rename", str(in, 2 + n));
329 if (read_only_) return denied("rename", path);
330 if (!fs_.stat(path.c_str(), e)) return Status::NotFound;
331 if (locks_->isBusy(path)) return busy("rename", path);
332 if (fs_.stat(path2.c_str(), e)) return Status::Exists;
333 if (fs_.rename(path.c_str(), path2.c_str())) return Status::Ok;
334 TINYBLEFS_LOGE("server", "rename %s to %s failed", path.c_str(), path2.c_str());
335 return Status::IoError;
336 }
337 }
338 return badRequest("unknown operation");
339 }
