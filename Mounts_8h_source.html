22 std::string mount(FileClient& client, const char* alias = nullptr) {
23 ServerInfo si;
24 if (client.info(si) != Status::Ok) {
25 TINYBLEFS_LOGW("mounts", "server does not answer to info request");
26 return "";
27 }
28 std::string base = sanitize(alias && *alias ? alias : si.name.c_str());
29 std::string name = base;
30 for (int i = 2; fs_.count(name); i++) name = base + "-" + std::to_string(i);
31 fs_.emplace(name, RemoteFS(client));
32 TINYBLEFS_LOGI("mounts", "mounted '%s' as /%s%s", si.name.c_str(), name.c_str(),
33 si.readOnly ? " (read-only)" : "");
34 return name;
35 }
37 bool unmount(const char* name) {
38 if (fs_.erase(name) > 0) return true;
39 TINYBLEFS_LOGW("mounts", "unmount: nothing is mounted as /%s", name ? name : "");
40 return false;
41 }

134 bool copy(const char* from, const char* to) {
135 RemoteFile src = open(from);
136 if (!src) {
137 TINYBLEFS_LOGW("mounts", "copy: cannot open source %s", from);
138 return false;
139 }
140 if (!src.isDirectory()) {
141 RemoteFile dst = open(to, "w");
142 if (!dst) {
143 TINYBLEFS_LOGW("mounts", "copy: cannot create destination %s", to);
144 return false;
145 }
146 return copyFile(src, dst);
147 }
148 // directory: create target (if missing) and copy the entries
149 std::string toDir = to;
150 std::string fromDir = std::string(from) + "/";
151 if ((toDir + "/").compare(0, fromDir.size(), fromDir) == 0) {
152 TINYBLEFS_LOGW("mounts", "copy: cannot copy %s into itself (%s)", from, to);
153 return false;
154 }
155 if (!exists(to) && !mkdir(to)) {
156 TINYBLEFS_LOGE("mounts", "copy: cannot create directory %s", to);
157 return false;
158 }
159 if (toDir.back() != '/') toDir += '/';
160 src.close();
161 std::vector<DirEntry> entries;
162 if (!list(from, entries)) {
163 TINYBLEFS_LOGW("mounts", "copy: cannot list %s", from);
164 return false;
165 }
166 for (const DirEntry& e : entries) {
167 std::string fromPath = std::string(from) + (from[strlen(from) - 1] == '/' ? "" : "/") + e.name;
168 if (!copy(fromPath.c_str(), (toDir + e.name).c_str())) return false;
169 }
170 return true;
171 }

174 static std::string sanitize(const char* n) {
175 std::string s;
176 for (; *n && s.size() < 16; n++) {
177 char c = *n;
178 bool ok = (c >= '0' && c <= '9') || (c >= 'a' && c <= 'z') ||
179 (c >= 'A' && c <= 'Z') || c == '-' || c == '_';
180 s += ok ? c : '_';
181 }
182 return s.empty() ? "server" : s;
183 }
184
186 RemoteFS* route(const char* path, std::string& rest) {
187 std::string p(path ? path : "");
188 if (p.empty() || p[0] != '/') {
189 TINYBLEFS_LOGW("mounts", "path '%s' must start with /<mount>", path ? path : "");
190 return nullptr;
191 }
192 size_t end = p.find('/', 1);
193 std::string name = p.substr(1, end == std::string::npos ? end : end - 1);
194 rest = end == std::string::npos ? "/" : p.substr(end);
195 RemoteFS* fs = get(name.c_str());
196 if (!fs) TINYBLEFS_LOGW("mounts", "no server mounted as /%s", name.c_str());
197 return fs;
198 }

200 std::map<std::string, RemoteFS> fs_;
