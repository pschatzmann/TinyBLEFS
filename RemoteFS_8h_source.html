26 size_t read(uint8_t* dst, size_t len) {
27 if (!alive() || s_->isDir) {
28 TINYBLEFS_LOGW("file", "read from a closed file or a directory");
29 return 0;
30 }
31 State& s = *s_;
32 size_t total = 0;
33 while (total < len) {
34 if (s.bufValid && s.pos >= s.bufStart &&
35 s.pos < s.bufStart + s.buf.size()) {
36 size_t off = s.pos - s.bufStart;
37 size_t n = std::min(len - total, s.buf.size() - off);
38 memcpy(dst + total, &s.buf[off], n);
39 total += n;
40 s.pos += (uint32_t)n;
41 } else {
42 std::vector<uint8_t> chunk;
43 Status st = s.c->read(s.handle, s.pos, s.c->chunkSize(), chunk);
44 if (st != Status::Ok) {
45 TINYBLEFS_LOGW("file", "read of %s at offset %u failed: %s", s.path.c_str(),
46 (unsigned)s.pos, statusName(st));
47 break;
48 }
49 if (chunk.empty()) break; // end of file
50 s.buf.swap(chunk);
51 s.bufStart = s.pos;
52 s.bufValid = true;
53 }
54 }
55 return total;
56 }

72 size_t write(const uint8_t* src, size_t len) {
73 if (!alive() || s_->isDir) {
74 TINYBLEFS_LOGW("file", "write to a closed file or a directory");
75 return 0;
76 }
77 State& s = *s_;
78 s.bufValid = false;
79 size_t total = 0;
80 while (total < len) {
81 size_t n = std::min<size_t>(len - total, s.c->chunkSize());
82 uint16_t written = 0;
83 Status st = s.c->write(s.handle, s.pos, src + total, n, written);
84 if (st != Status::Ok) {
85 TINYBLEFS_LOGW("file", "write to %s at offset %u failed: %s", s.path.c_str(),
86 (unsigned)s.pos, statusName(st));
87 break;
88 }
89 total += written;
90 s.pos += written;
91 if (s.pos > s.size) s.size = s.pos;
92 if (written < n) {
93 TINYBLEFS_LOGW("file", "short write to %s: %u of %u bytes", s.path.c_str(),
94 (unsigned)written, (unsigned)n);
95 break;
96 }
97 }
98 return total;
99 }

128 RemoteFile openNextFile(const char* mode = "r") {
129 if (!isDirectory()) return RemoteFile();

132 if (idx < s.dirBufStart || idx >= s.dirBufStart + s.dirBuf.size()) {
133 // the entry is not in the buffered batch: fetch the batch that starts here

198 static uint8_t modeFlags(const char* mode) {
199 if (!mode || mode[0] == 'r') {
200 return (mode && strchr(mode, '+')) ? (kRead | kWrite) : kRead;
201 }
202 if (mode[0] == 'w') return kWrite | kCreate | kTrunc;
203 if (mode[0] == 'a') return kWrite | kCreate | kAppend;
204 return kRead;
205 }

207 static RemoteFile make(FileClient& c, const std::string& path, uint8_t flags,
208 Status* status = nullptr) {
209 uint8_t handle = 0;
210 uint32_t size = 0;
211 Status st = c.open(path, flags, handle, size);
212 if (status) *status = st;
213 if (st == Status::Busy) {
214 // with openWait() the caller retries and reports; a plain open reports
215 if (status) TINYBLEFS_LOGD("file", "%s is busy", path.c_str());
216 else TINYBLEFS_LOGW("file", "cannot open %s: busy (open for writing by another client)", path.c_str());
217 return RemoteFile();
218 }
219 if (st == Status::NotFound) {
220 TINYBLEFS_LOGD("file", "%s does not exist", path.c_str());
221 return RemoteFile();
222 }
223 if (st != Status::Ok && st != Status::IsDirectory) {
224 TINYBLEFS_LOGW("file", "cannot open %s: %s", path.c_str(), statusName(st));
225 return RemoteFile();
226 }
227 auto s = std::make_shared<State>();
228 s->c = &c;
229 s->gen = c.generation();
230 s->path = path;
231 size_t p = path.find_last_of('/');
232 s->name = p == std::string::npos ? path : path.substr(p + 1);
233 if (st == Status::IsDirectory) {
234 s->isDir = true;
235 } else {
236 s->handle = handle;
237 s->size = size;
238 s->pos = (flags & kAppend) ? size : 0;
239 }
240 s->valid = true;
241 return RemoteFile(s);
242 }

244 std::shared_ptr<State> s_;

249inline bool copyFile(RemoteFile& src, RemoteFile& dst, size_t bufSize = 512) {
250 if (!src || !dst || src.isDirectory() || dst.isDirectory()) {
251 TINYBLEFS_LOGW("copy", "source or destination is not an open file");
252 return false;
253 }
254 std::vector<uint8_t> buf(bufSize);
255 while (true) {
256 size_t n = src.read(buf.data(), buf.size());
257 if (n == 0) break;
258 if (dst.write(buf.data(), n) != n) {
259 TINYBLEFS_LOGE("copy", "write to %s failed", dst.path());
260 return false;
261 }
262 }
263 return true;

280 uint32_t timeoutMs = 5000, uint32_t retryMs = 100) {
281 uint32_t start = nowMs();

283 Status st = Status::Ok;
