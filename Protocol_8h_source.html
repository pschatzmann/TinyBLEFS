2// Wire protocol of TinyBLEFS: opcodes, status codes and message framing.
3//
4// Message frame (little endian):
5// [op:1][reqId:1][status:1][payloadLen:2][payload...]
6// Requests have status 0; responses echo reqId and set op |= kResponseFlag.
7// A message may be split over several transport packets (MTU sized); BLE
8// delivers packets reliably and in order so the length field is all that is
9// needed to reassemble.
10//
11// All file access is offset based (stateless cursor), so a retried READ or
12// WRITE is idempotent.

156inline std::vector<uint8_t> encode(const Message& m) {
157 std::vector<uint8_t> out;
158 out.reserve(kHeaderSize + m.payload.size());
159 out.push_back(m.op);
160 out.push_back(m.reqId);
161 out.push_back(m.status);
162 put16(out, (uint16_t)m.payload.size());
163 out.insert(out.end(), m.payload.begin(), m.payload.end());
164 return out;
165}

168class Reassembler {
169 public:
171 void feed(const uint8_t* data, size_t len) {
172 buf_.insert(buf_.end(), data, data + len);
173 }
175 bool next(Message& m) {
176 if (buf_.size() < kHeaderSize) return false;
177 size_t len = get16(&buf_[3]);
178 if (buf_.size() < kHeaderSize + len) return false;
179 m.op = buf_[0];
180 m.reqId = buf_[1];
181 m.status = buf_[2];
182 m.payload.assign(buf_.begin() + kHeaderSize,
183 buf_.begin() + kHeaderSize + len);
184 buf_.erase(buf_.begin(), buf_.begin() + kHeaderSize + len);
185 return true;
186 }
188 void reset() { buf_.clear(); }
189
190 private:
191 std::vector<uint8_t> buf_;
192};
