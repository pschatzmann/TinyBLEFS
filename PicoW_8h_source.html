2// Transport adapters for the Raspberry Pi Pico W / Pico 2 W using the BLE
3// library of the arduino-pico core (BTstack based).
4// Select Tools -> IP/Bluetooth Stack -> "IPv4 + Bluetooth" in the IDE
5// (fqbn option ipbtstack=ipv4btcble).
6//
7// Server: tinyblefs::PicoWServerTransport (several clients: see Session.h)
8// Client: tinyblefs::PicoWClientTransport -- the core's BLE library has one
9// global BLEClient, so only ONE remote server can be connected.
10//
11// The core's BLECharacteristic::setValue() notification cannot be used to send
12// a sequence of packets (it keeps a single pending send request that is
13// overwritten by the next call). The server therefore sends its notifications
14// directly with BTstack's att_server_notify(), waiting until the ACL buffers
15// are free. This uses protected members of BLECharacteristic (a subclass).

27// The Pico accepts writes of up to ATT MTU 517 - 3 bytes: the receive queue must

31#define TINYBLEFS_PICO_RX_PACKET 514

46 bool sendPacket(uint16_t handle, const uint8_t* data, size_t len,
47 uint32_t timeoutMs = 2000) {
48 uint32_t start = millis();
49 while (millis() - start < timeoutMs) {
50 {
51 BluetoothLock lock;
52 if (att_server_can_send_packet_now(handle)) {
53 uint8_t err = att_server_notify(handle, _valueHandle, (uint8_t*)data, len);
54 if (err != ERROR_CODE_SUCCESS)
55 TINYBLEFS_LOGE("picow", "att_server_notify failed (error 0x%02x)", err);
56 return err == ERROR_CODE_SUCCESS;
57 }
58 }
59 delay(1);
60 }
61 TINYBLEFS_LOGW("picow", "notification not sent within %u ms: Bluetooth buffers full",
62 (unsigned)timeoutMs);
63 return false;
64 }
66 bool subscribed() const { return _notificationEnabled; }
68 void setSubscribed(bool on) { _notificationEnabled = on; }
69
70 protected:
71 // The core would switch notifications off when ANY client disconnects; the
72 // transport does that when the last one has gone.
74 void disconnected() override {}
75};

120 // ---- Transport -------------------------------------------------------
121 size_t mtu() const override {
122 uint16_t h[kMaxConnections];
123 size_t n = snapshot(h);
124 size_t best = TINYBLEFS_PICO_RX_PACKET;
125 if (n == 0) return 20;
126 for (size_t i = 0; i < n; i++) {
127 uint16_t m = att_server_get_mtu(h[i]);
128 size_t v = m > 23 ? m - 3 : 20;
129 if (v < best) best = v;
130 }
131 return best;
132 }
135 bool send(const uint8_t* data, size_t len) override {
136 uint16_t h[kMaxConnections];
137 size_t n = snapshot(h);
138 if (n == 0 || !tx_.subscribed()) {
139 TINYBLEFS_LOGW("picow", "cannot notify: no client connected or not subscribed");
140 return false;
141 }
142 bool ok = false;
143 for (size_t i = 0; i < n; i++) ok = tx_.sendPacket(h[i], data, len) || ok;
144 TINYBLEFS_LOGD("picow", "sent packet of %u bytes (ATT MTU %u)", (unsigned)len,
145 (unsigned)att_server_get_mtu(h[0]));
146 return ok;
147 }
148 size_t receive(uint8_t* buf, size_t max) override {
149 if (clear_pending_) {
150 clear_pending_ = false;
151 q_.clear();
152 }
153 uint32_t n = q_.takeDropped();
154 if (n) TINYBLEFS_LOGW("picow", "%u received packet(s) dropped (queue full or packet too large)", (unsigned)n);
155 size_t len = q_.pop(buf, max);
156 if (len) TINYBLEFS_LOGD("picow", "received packet of %u bytes", (unsigned)len);
157 return len;
158 }
159 bool connected() const override { return count_ > 0; }

177 // HCI events (Bluetooth context)
178 static void onHci(uint8_t type, uint16_t, uint8_t* packet, uint16_t) {
179 PicoWServerTransport* t = self();
180 if (!t || type != HCI_EVENT_PACKET) return;
181 switch (hci_event_packet_get_type(packet)) {
182 case HCI_EVENT_META_GAP:
183 if (hci_event_gap_meta_get_subevent_code(packet) == GAP_SUBEVENT_LE_CONNECTION_COMPLETE)
184 t->added(gap_subevent_le_connection_complete_get_connection_handle(packet));
185 break;
186 case HCI_EVENT_DISCONNECTION_COMPLETE:
187 t->removed(hci_event_disconnection_complete_get_connection_handle(packet));
188 break;
189 default:
190 break;
191 }
192 }
193 void added(uint16_t handle) {
194 if (count_ == 0) clear_pending_ = true;
195 if (count_ < max_) handles_[count_++] = handle;
196 gap_request_connection_parameter_update(handle, conn_min_, conn_max_, 0, 200);
197 if (count_ < max_) gap_advertisements_enable(1); // room for more clients
198 }
199 void removed(uint16_t handle) {
200 for (size_t i = 0; i < count_; i++) {
201 if (handles_[i] != handle) continue;
202 handles_[i] = handles_[--count_];
203 if (count_ == 0) tx_.setSubscribed(false);
204 gap_advertisements_enable(1); // BTstack stops advertising while connected
205 return;
206 }
207 }
208
209 BLEService service_;
210 BLECharacteristic rx_;
211 detail::PicoWNotifyCharacteristic tx_;
212 RingQueueT<16, TINYBLEFS_PICO_RX_PACKET> q_;
213 btstack_packet_callback_registration_t hci_reg_;
214 uint16_t handles_[kMaxConnections] = {};
215 volatile size_t count_ = 0;
216 size_t max_ = 1;
217 uint16_t conn_min_ = 6, conn_max_ = 6;
218 volatile bool clear_pending_ = false;

249 bool connect(BLEAdvertising& device, uint32_t timeoutSec = 10) {
250 disconnect();
251 BLEClient* c = BLE.client();
252 if (!c->connect(device, timeoutSec)) {
253 TINYBLEFS_LOGW("picow", "connect failed: cannot connect to the device");
254 return false;
255 }
256 BLERemoteService* svc = c->service(BLEUUID(kServiceUUID));
257 if (!svc) return fail("device has no TinyBLEFS service");
258 rx_ = svc->characteristic(BLEUUID(kRxUUID));
259 BLERemoteCharacteristic* tx = svc->characteristic(BLEUUID(kTxUUID));
260 if (!rx_ || !tx) return fail("TinyBLEFS characteristics not found");
261 if (!rx_->canWrite()) return fail("the TinyBLEFS write characteristic is not writable");
262 self() = this;
263 q_.clear();
264 tx->onNotify(&PicoWClientTransport::onNotify);
265 if (!tx->enableNotifications()) return fail("cannot enable notifications");
266 return true;
267 }
268
270 void disconnect() {
271 if (rx_) {
272 BLE.client()->disconnect();
273 rx_ = nullptr;
274 }
275 if (self() == this) self() = nullptr;
276 }
