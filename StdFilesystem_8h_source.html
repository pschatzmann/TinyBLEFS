24 int read(uint32_t offset, uint8_t* buf, size_t len) override {
25 f_.clear();
26 f_.seekg(offset);
27 f_.read((char*)buf, len);
28 int n = (int)f_.gcount();

33 f_.clear();

37 f_.clear();
38 f_.seekp(offset);

43 return -1;
44 }
45 return (int)len;
46 }
47 uint32_t size() override {
48 f_.clear();
49 f_.seekg(0, std::ios::end);
50 return (uint32_t)f_.tellg();
51 }

58 public:
59 FsFile* open(const char* path, uint8_t flags) override {
60 namespace fs = std::filesystem;
61 std::error_code ec;
62 bool exists = fs::is_regular_file(path, ec);
63 bool write = flags & (kWrite | kAppend | kCreate | kTrunc);
64 std::ios::openmode mode = std::ios::in;
65 if (write) {
66 if (!exists) {
67 if (!(flags & (kCreate | kTrunc))) return nullptr;
68 std::ofstream create(path, std::ios::binary); // create empty file
69 if (!create) {
70 TINYBLEFS_LOGE("stdfs", "cannot create %s: %s", path, strerror(errno));
71 return nullptr;
72 }
73 } else if (flags & kTrunc) {
74 std::ofstream trunc(path, std::ios::binary | std::ios::trunc);
75 if (!trunc) {
76 TINYBLEFS_LOGE("stdfs", "cannot truncate %s: %s", path, strerror(errno));
77 return nullptr;
78 }
79 }
80 mode |= std::ios::out;
81 } else if (!exists) {
82 return nullptr;
83 }
84 StdFile* f = new StdFile(path, mode);
85 if (!f->isOpen()) {
86 TINYBLEFS_LOGE("stdfs", "cannot open %s: %s", path, strerror(errno));
87 delete f;
88 return nullptr;
89 }
90 return f;
91 }

93 bool stat(const char* path, FsEntry& e) override {
94 namespace fs = std::filesystem;
95 std::error_code ec;
96 auto st = fs::status(path, ec);
97 if (ec || !fs::exists(st)) {
98 if (ec) TINYBLEFS_LOGD("stdfs", "stat %s: %s", path, ec.message().c_str());
99 return false;
100 }
101 e.name = fs::path(path).filename().string();
102 e.isDir = fs::is_directory(st);
103 e.size = e.isDir ? 0 : (uint32_t)fs::file_size(path, ec);
104 return true;

108 namespace fs = std::filesystem;
