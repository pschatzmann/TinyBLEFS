2// Server backend for the TinyFATFS library (FatFs by ChaN as header-only
3// Arduino library): share an SD card (SPI / SDMMC), a RAM disk, a USB stick,
4// ... over BLE.
5// https://github.com/pschatzmann/TinyFATFS

7// RamIO drive{200, 512}; // or the SD card driver
8// TinyFatBackend backend(SD); // SD: the global object of TinyFATFS
9// FileServer server(backend, transport, "sd");
10// void setup() { SD.begin(drive); transport.begin("TinyFS SD"); }
11//
12// SD.begin() must succeed before the first client connects. Data is written to
13// the medium when the client closes the file.
14//
15// If FF_FS_LOCK is used in ffconf.h of TinyFATFS it must be at least the

34 FsFile* open(const char* path, uint8_t flags) override {
35 bool write = flags & (kWrite | kAppend | kCreate | kTrunc);
36 uint8_t mode = FA_READ;
37 if (write) {
38 mode |= FA_WRITE;
39 if (flags & kTrunc) mode |= FA_CREATE_ALWAYS; // create or truncate
40 else if (flags & kCreate) mode |= FA_OPEN_ALWAYS; // create, keep content
41 else mode |= FA_OPEN_EXISTING;
42 }
43 std::unique_ptr<TinyFatFile> f(new TinyFatFile(sd_, path, mode));
44 if (!f->isOpen()) {
45 TINYBLEFS_LOGE("tinyfat", "cannot open %s (mode 0x%02x)", path, mode);
46 return nullptr;
47 }
48 return f.release();
49 }
50
52 bool stat(const char* path, FsEntry& e) override {
53 if (isRoot(path)) { // FatFs has no FILINFO for the root directory
54 e.name.clear();
55 e.isDir = true;
56 e.size = 0;
57 return true;
58 }
59 fatfs::FILINFO info;
60 fatfs::FRESULT rc = sd_.getFatFs()->f_stat(path, &info);
61 if (rc != fatfs::FR_OK) {
62 TINYBLEFS_LOGD("tinyfat", "stat %s: FatFs error %d", path, (int)rc);
63 return false;
64 }
65 fill(info, e);
66 return true;
67 }
68
69 bool list(const char* path, uint32_t index, FsEntry& e) override {
70 fatfs::FatFs* fs = sd_.getFatFs();
71 fatfs::DIR dir;
72 fatfs::FRESULT rc = fs->f_opendir(&dir, path);
73 if (rc != fatfs::FR_OK) {
74 TINYBLEFS_LOGW("tinyfat", "list %s: FatFs error %d", path, (int)rc);
75 return false;
76 }
77 bool found = false;
78 fatfs::FILINFO info;
79 for (uint32_t i = 0;; i++) {
80 if (fs->f_readdir(&dir, &info) != fatfs::FR_OK || info.fname[0] == 0) break;
81 if (i == index) {
82 fill(info, e);
83 found = true;
84 break;
85 }
86 }
87 fs->f_closedir(&dir);
88 return found;
89 }
