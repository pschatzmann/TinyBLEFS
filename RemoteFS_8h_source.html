29 size_t read(uint8_t* dst, size_t len) {
30 if (!alive() || s_->isDir) {
31 TINYBLEFS_LOGW("file", "read from a closed file or a directory");
32 return 0;
33 }
34 State& s = *s_;
35 size_t total = 0;
36 while (total < len) {
37 if (s.bufValid && s.pos >= s.bufStart &&
38 s.pos < s.bufStart + s.buf.size()) {
39 size_t off = s.pos - s.bufStart;
40 size_t n = std::min(len - total, s.buf.size() - off);
41 memcpy(dst + total, &s.buf[off], n);
42 total += n;
43 s.pos += (uint32_t)n;
44 } else {
45 std::vector<uint8_t> chunk;
46 Status st = s.c->read(s.handle, s.pos, s.c->chunkSize(), chunk);
47 if (st != Status::Ok) {
48 TINYBLEFS_LOGW("file", "read of %s at offset %u failed: %s", s.path.c_str(),
49 (unsigned)s.pos, statusName(st));
50 break;
51 }
52 if (chunk.empty()) break; // end of file
53 s.buf.swap(chunk);
54 s.bufStart = s.pos;
55 s.bufValid = true;
56 }
57 }
58 return total;
59 }

79 size_t write(const uint8_t* src, size_t len) {
80 if (!alive() || s_->isDir) {
81 TINYBLEFS_LOGW("file", "write to a closed file or a directory");
82 return 0;
83 }
84 State& s = *s_;
85 s.bufValid = false;
86 size_t total = 0;
87 while (total < len) {
88 size_t n = std::min<size_t>(len - total, s.c->chunkSize());
89 uint16_t written = 0;
90 Status st = s.c->write(s.handle, s.pos, src + total, n, written);
91 if (st != Status::Ok) {
92 TINYBLEFS_LOGW("file", "write to %s at offset %u failed: %s", s.path.c_str(),
93 (unsigned)s.pos, statusName(st));
94 break;
95 }
96 total += written;
97 s.pos += written;
98 if (s.pos > s.size) s.size = s.pos;
99 if (written < n) {
100 TINYBLEFS_LOGW("file", "short write to %s: %u of %u bytes", s.path.c_str(),
101 (unsigned)written, (unsigned)n);
102 break;
103 }
104 }
105 return total;
106 }

146 RemoteFile openNextFile(const char* mode = "r") {
147 if (!isDirectory()) return RemoteFile();
148 State& s = *s_;
149 size_t idx = s.dirIndex;
150 if (idx < s.dirBufStart || idx >= s.dirBufStart + s.dirBuf.size()) {
151 // the entry is not in the buffered batch: fetch the batch that starts here
152 if (s.dirEnd && idx >= s.dirBufStart + s.dirBuf.size()) return RemoteFile();
153 Status st = s.c->listBatch(s.path, s.dirIndex, s.c->chunkSize(), s.dirBuf, s.dirEnd);
154 s.dirBufStart = s.dirIndex;
155 if (st != Status::Ok) {
156 TINYBLEFS_LOGW("file", "listing of %s failed: %s", s.path.c_str(), statusName(st));
157 s.dirBuf.clear();

159 }
160 if (s.dirBuf.empty()) return RemoteFile();
161 }
162 const DirEntry& e = s.dirBuf[idx - s.dirBufStart];
163 s.dirIndex++;

218 static uint8_t modeFlags(const char* mode) {
219 if (!mode || mode[0] == 'r') {
220 return (mode && strchr(mode, '+')) ? (kRead | kWrite) : kRead;
221 }
222 if (mode[0] == 'w') return kWrite | kCreate | kTrunc;
223 if (mode[0] == 'a') return kWrite | kCreate | kAppend;
224 return kRead;
225 }

227 static RemoteFile make(FileClient& c, const std::string& path, uint8_t flags,
228 Status* status = nullptr) {
229 uint8_t handle = 0;
230 uint32_t size = 0;
231 Status st = c.open(path, flags, handle, size);
232 if (status) *status = st;
233 if (st == Status::Busy) {
234 // with openWait() the caller retries and reports; a plain open reports
235 if (status) TINYBLEFS_LOGD("file", "%s is busy", path.c_str());
236 else TINYBLEFS_LOGW("file", "cannot open %s: busy (open for writing by another client)", path.c_str());
237 return RemoteFile();
238 }
239 if (st == Status::NotFound) {
240 TINYBLEFS_LOGD("file", "%s does not exist", path.c_str());
241 return RemoteFile();
242 }
243 if (st != Status::Ok && st != Status::IsDirectory) {
244 TINYBLEFS_LOGW("file", "cannot open %s: %s", path.c_str(), statusName(st));
245 return RemoteFile();
246 }
247 auto s = std::make_shared<State>();
248 s->c = &c;
249 s->gen = c.generation();
250 s->path = path;
251 size_t p = path.find_last_of('/');
252 s->name = p == std::string::npos ? path : path.substr(p + 1);
253 if (st == Status::IsDirectory) {
254 s->isDir = true;
255 } else {
256 s->handle = handle;
257 s->size = size;
258 s->pos = (flags & kAppend) ? size : 0;
259 }
260 s->valid = true;
261 return RemoteFile(s);
262 }
263
264 std::shared_ptr<State> s_;

269inline bool copyFile(RemoteFile& src, RemoteFile& dst, size_t bufSize = 512) {
270 if (!src || !dst || src.isDirectory() || dst.isDirectory()) {
271 TINYBLEFS_LOGW("copy", "source or destination is not an open file");
272 return false;
273 }
274 std::vector<uint8_t> buf(bufSize);
275 while (true) {
276 size_t n = src.read(buf.data(), buf.size());
277 if (n == 0) break;
278 if (dst.write(buf.data(), n) != n) {
279 TINYBLEFS_LOGE("copy", "write to %s failed", dst.path());
280 return false;
281 }
282 }
283 // read() returns 0 both at the end and on a failed request
284 if (src.position() < src.size()) {
285 TINYBLEFS_LOGE("copy", "read of %s failed at %u of %u", src.path(),
286 (unsigned)src.position(), (unsigned)src.size());
287 return false;
288 }
289 return true;

306 RemoteFile openWait(const char* path, const char* mode = "r",
307 uint32_t timeoutMs = 5000, uint32_t retryMs = 100) {
308 uint32_t start = nowMs();
309 while (true) {
310 Status st = Status::Ok;
311 RemoteFile f = RemoteFile::make(*c_, path, RemoteFile::modeFlags(mode), &st);
312 if (f || st != Status::Busy) return f;
313 if (nowMs() - start >= timeoutMs) {
314 TINYBLEFS_LOGW("fs", "%s is still busy after %u ms", path, (unsigned)timeoutMs);
315 return f;
316 }
317 sleepMs(retryMs);
318 }
