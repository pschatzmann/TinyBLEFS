132 Status listBatch(const std::string& path, uint16_t index, uint16_t maxBytes,
133 std::vector<DirEntry>& entries, bool& end) {
134 entries.clear();
135 end = false;
136 if (batch_list_) {
137 std::vector<uint8_t> in, out;
138 put16(in, index);
139 put16(in, maxBytes);
140 putStr(in, path);
141 Status st = request(Op::ListBatch, in, &out, 1);
142 if (st == Status::Ok) {
143 if (out.empty()) return malformed(__func__);
144 size_t pos = 1;
145 while (pos < out.size()) {
146 if (pos + 6 > out.size() || pos + 6 + out[pos + 5] > out.size()) return malformed(__func__);
147 DirEntry e;
148 e.isDir = out[pos] != 0;
149 e.size = get32(&out[pos + 1]);
150 e.name.assign((const char*)&out[pos + 6], out[pos + 5]);
151 pos += 6 + out[pos + 5];
152 entries.push_back(std::move(e));
153 }
154 end = (out[0] & kListEnd) != 0;
155 return Status::Ok;
156 }
157 if (st != Status::BadRequest) return st;
158 // An older server answers "bad request" to the unknown operation (a bad
159 // path does too): ask entry by entry, and remember it if that works.
160 Status fb = listEntryByEntry(path, index, maxBytes, entries, end);
161 if (fb == Status::Ok) {
162 TINYBLEFS_LOGI("client", "server has no batched listing: listing entry by entry");
163 batch_list_ = false;
164 }
165 return fb;
166 }
167 return listEntryByEntry(path, index, maxBytes, entries, end);
168 }

186 // ---- low level -----------------------------------------------------
189 Status request(Op op, const std::vector<uint8_t>& payload,
190 std::vector<uint8_t>* response = nullptr, int retries = 0) {
191 request_count_++;
192 for (int attempt = 0; attempt <= retries; attempt++) {
193 Message req;
194 req.op = (uint8_t)op;
195 req.reqId = ++req_id_;
196 req.payload = payload;
197 if (!sendMessage(t_, req)) {
198 TINYBLEFS_LOGE("client", "cannot send %s request", opName(req.op));
199 if (attempt < retries && t_.connected()) continue; // transient: try again
200 return Status::SendFailed;
201 }
202 uint32_t start = nowMs();
203 while (nowMs() - start < timeout_) {
204 if (idle_) idle_(idle_ctx_);
205 t_.poll();
206 Message rsp;
207 bool got = false;
208 while (rx_.poll(t_, rsp)) {
209 got = true;
210 if (rsp.reqId == req.reqId && rsp.op == (req.op | kResponseFlag)) {
211 Status st = (Status)rsp.status;
212 if (st != Status::Ok && st != Status::NotFound &&
213 st != Status::EndOfList && st != Status::IsDirectory &&
214 st != Status::Busy)
215 TINYBLEFS_LOGW("client", "%s request failed: %s", opName(req.op), statusName(st));
216 if (response) *response = std::move(rsp.payload);
217 return st;
218 }
219 TINYBLEFS_LOGD("client", "ignoring stale response %u", (unsigned)rsp.reqId);
220 }
221 if (!got) yieldCpu();
222 if (!t_.connected()) {
223 TINYBLEFS_LOGE("client", "link lost while waiting for %s response", opName(req.op));
224 return Status::SendFailed;
225 }
226 }
227 TINYBLEFS_LOGW("client", "%s request timed out after %u ms (attempt %d of %d)",
228 opName(req.op), (unsigned)timeout_, attempt + 1, retries + 1);
229 }
230 return Status::Timeout;
231 }

234 Status listEntryByEntry(const std::string& path, uint16_t index, uint16_t maxBytes,
235 std::vector<DirEntry>& entries, bool& end) {
236 size_t used = 1;
237 while (true) {
238 DirEntry e;
239 Status st = list(path, (uint16_t)(index + entries.size()), e.isDir, e.size, e.name);
240 if (st == Status::EndOfList) {
241 end = true;
242 return Status::Ok;
243 }
244 if (st != Status::Ok) return st;
245 used += 6 + e.name.size();
246 entries.push_back(std::move(e));
247 if (used >= maxBytes) return Status::Ok;
248 }
249 }
