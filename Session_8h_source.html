2// Sessions: every client works in its own session on the server (own open
3// files, own message buffer), so that several clients can share one BLE link,
4// also on BLE stacks that can neither tell which client sent a packet nor
5// address a reply to one client (ArduinoBLE, Pico W). The session id is the
6// first byte of every packet.
7//
8// Applications do not use this file directly: FileServer(..., maxSessions) and
9// FileClient handle the sessions.
10//
11// All connected clients receive all replies (the stack broadcasts them), so
12// they must be trusted with each other's data.
13//
14// Packet format: [sessionId:1][data...]. Session id 0 is a control packet:
15// [0][1][nonce:4] Hello client -> server
16// [0][2][nonce:4][id:1] Welcome server -> client
17// [0][3][nonce:4] Full server -> client (no free session)
18// [0][4][id:1] Expired server -> client (unknown session)
19// [0][5][id:1] KeepAlive client -> server

58 bool join() {
59 uint32_t nonce = randomU32();
60 uint32_t start = nowMs(), lastTry = 0;
61 bool sent = false;
62 sid_ = 0;
63 while (nowMs() - start < timeout_) {
64 if (!raw_.connected()) {
65 TINYBLEFS_LOGW("session", "join: link is down");
66 return false;
67 }
68 if (!sent || nowMs() - lastTry > 700) { // the hello may get lost
69 std::vector<uint8_t> h{0, kSessionHello};
70 put32(h, nonce);
71 if (!raw_.send(h.data(), h.size())) TINYBLEFS_LOGW("session", "cannot send hello");
72 lastTry = nowMs();
73 sent = true;
74 }
75 if (idle_) idle_(idle_ctx_);
76 raw_.poll();
77 size_t n;
78 while ((n = raw_.receive(buf_.data(), buf_.size())) > 0) {
79 if (n >= 6 && buf_[0] == 0 && get32(&buf_[2]) == nonce) {
80 if (buf_[1] == kSessionWelcome && n >= 7) {
81 sid_ = buf_[6];
82 expired_ = false;
83 gen_++; // files of an earlier session are gone
84 TINYBLEFS_LOGI("session", "joined as session %u", (unsigned)sid_);
85 return true;
86 }
87 if (buf_[1] == kSessionFull) {
88 TINYBLEFS_LOGW("session", "the server has no free session");
89 return false;
90 }
91 }
92 }
93 yieldCpu();
94 }
95 TINYBLEFS_LOGW("session", "no answer to hello after %u ms", (unsigned)timeout_);
96 return false;
97 }
98
101 void keepAlive() {
102 if (!sid_ || expired_) return;
103 uint8_t p[3] = {0, kSessionKeepAlive, sid_};
104 raw_.send(p, sizeof(p));
105 }

130 }
131 size_t receive(uint8_t* out, size_t max) override {
132 size_t n;
133 while ((n = raw_.receive(buf_.data(), buf_.size())) > 0) {
134 if (buf_[0] == 0) {
135 if (n >= 3 && buf_[1] == kSessionExpired && buf_[2] == sid_ && sid_) {
136 TINYBLEFS_LOGW("session", "session %u was closed by the server (idle too long?)",
137 (unsigned)sid_);
138 expired_ = true;
139 }
140 continue;
141 }
142 if (buf_[0] != sid_ || n < 2) continue; // reply for another client
143 size_t len = n - 1 < max ? n - 1 : max;
144 memcpy(out, buf_.data() + 1, len);
145 return len;
146 }
147 return 0;
148 }
151 bool connected() const override { return raw_.connected() && !expired_; }
152 void poll() override { raw_.poll(); }
