37inline bool sendMessage(Transport& t, const Message& m) {
38 std::vector<uint8_t> data = encode(m);
39 size_t mtu = t.mtu();
40 if (mtu == 0) {
41 TINYBLEFS_LOGE("transport", "packet size (mtu) is 0: cannot send");
42 return false;
43 }
44 if (m.payload.size() > kMaxPayload) {
45 TINYBLEFS_LOGE("transport", "message of %u bytes is too large",
46 (unsigned)m.payload.size());
47 return false;
48 }
49 for (size_t off = 0; off < data.size(); off += mtu) {
50 size_t n = std::min(mtu, data.size() - off);
51 if (!t.send(&data[off], n)) {
52 TINYBLEFS_LOGW("transport", "sending packet at offset %u of %u failed",
53 (unsigned)off, (unsigned)data.size());
54 return false;
55 }
56 }
57 return true;
58}

61class Receiver {

63 bool poll(Transport& t, Message& m) {
64 // the buffer is a member, not a local: small MCUs have only ~1-4 KB of stack
65 if (tmp_.empty()) tmp_.resize(kPacketBuffer);
66 while (true) {
67 if (asm_.next(m)) return true;
68 size_t n = t.receive(tmp_.data(), tmp_.size());
69 if (n == 0) return false;
70 asm_.feed(tmp_.data(), n);
71 }
