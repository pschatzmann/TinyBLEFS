39class LockedQueue {
40 public:
42 void push(const uint8_t* d, size_t n) {
43 std::lock_guard<std::mutex> g(m_);
44 if (q_.size() < 64) q_.emplace_back(d, d + n);
45 else dropped_++;
46 }
48 size_t pop(uint8_t* buf, size_t max) {
49 std::lock_guard<std::mutex> g(m_);
50 if (q_.empty()) return 0;
51 size_t n = std::min(max, q_.front().size());
52 memcpy(buf, q_.front().data(), n);
53 q_.pop_front();
54 return n;
55 }
57 void clear() {
58 std::lock_guard<std::mutex> g(m_);
59 q_.clear();
60 }
62 uint32_t takeDropped() {
63 std::lock_guard<std::mutex> g(m_);
64 uint32_t n = dropped_;
65 dropped_ = 0;
66 return n;
67 }
68
69 private:
70 std::mutex m_;
71 uint32_t dropped_ = 0;
72 std::deque<std::vector<uint8_t>> q_;
73};

78inline bool selectAdapter(SimpleBLE::Adapter& out) {
79 auto adapters = SimpleBLE::Adapter::get_adapters();
80 if (adapters.empty()) {
81 TINYBLEFS_LOGE("simpleble", "no Bluetooth adapter found (is bluetoothd running?)");
82 return false;
83 }
84 const char* want = getenv("TINYBLEFS_ADAPTER");
85 if (want && *want) {
86 for (auto& a : adapters) {
87 if (a.identifier() == want || strcasecmp(a.address().c_str(), want) == 0) {
88 out = a;
89 return true;
90 }
91 }
92 TINYBLEFS_LOGE("simpleble", "adapter '%s' (TINYBLEFS_ADAPTER) not found", want);
93 return false;
94 }
95 for (auto& a : adapters) {
96 if (a.is_powered()) {
97 out = a;
98 return true;
99 }
100 }
101 out = adapters[0];
102 return true;
103}

114 static std::vector<SimpleBLE::Peripheral> scan(
115 int durationMs = 5000, const std::string& namePrefix = "") {
116 std::vector<SimpleBLE::Peripheral> found;
117 std::vector<SimpleBLE::Peripheral> results;
118 try {
119 SimpleBLE::Adapter adapter;
120 if (!detail::selectAdapter(adapter)) return found;
121 adapter.scan_for(durationMs);
122 results = adapter.scan_get_results();
123 } catch (const std::exception& e) {
124 TINYBLEFS_LOGE("simpleble", "scan failed: %s", e.what());
125 return found;
126 }
127 for (auto& p : results) {
128 bool tinyblefs = false;
129 for (auto& s : p.services())
130 if (equalsNoCase(s.uuid(), kServiceUUID)) tinyblefs = true;
131 if (!tinyblefs) continue;
132 if (!namePrefix.empty() && p.identifier().compare(0, namePrefix.size(), namePrefix) != 0)
133 continue;
134 found.push_back(p);
135 }
136 return found;
137 }

140 bool connect(SimpleBLE::Peripheral peripheral) {
141 disconnect();
142 try {
143 peripheral_ = peripheral;
144 peripheral_.connect();
145 has_peripheral_ = true;
146 TINYBLEFS_LOGI("simpleble", "connected, packets of %u bytes (ATT MTU %u)",
147 (unsigned)mtu(), (unsigned)peripheral_.mtu());
148 // after connect() BlueZ has resolved the services and negotiated the MTU
149 peripheral_.notify(kServiceUUID, kTxUUID, [this](SimpleBLE::ByteArray payload) {
150 q_.push((const uint8_t*)payload.data(), payload.size());
151 });
152 return true;
153 } catch (const std::exception& e) {
154 TINYBLEFS_LOGW("simpleble", "connect failed: %s", e.what());
155 disconnect();
156 return false;
157 }
158 }

173 // ---- Transport -------------------------------------------------------
174 size_t mtu() const override {
175 if (!has_peripheral_) return 20;
176 try {
177 uint16_t m = const_cast<SimpleBLE::Peripheral&>(peripheral_).mtu();
178 // ATT allows at most 517; BlueZ reports nonsense (65533) for some devices
179 if (m > 517) m = 517;
180 return m > 23 ? m - 3 : 20;
181 } catch (const std::exception& e) {
182 TINYBLEFS_LOGD("simpleble", "cannot read the MTU: %s", e.what());
183 return 20;
184 }
185 }
186 bool send(const uint8_t* data, size_t len) override {
187 if (!has_peripheral_) return false;
188 try {
189 peripheral_.write_command(kServiceUUID, kRxUUID,
190 SimpleBLE::ByteArray((const char*)data, len));
191 return true;
192 } catch (const std::exception& e) {
193 TINYBLEFS_LOGW("simpleble", "write of %u bytes failed: %s", (unsigned)len, e.what());
194 return false;
195 }
196 }
197 size_t receive(uint8_t* buf, size_t max) override {
198 uint32_t d = q_.takeDropped();
199 if (d) TINYBLEFS_LOGW("simpleble", "%u received packet(s) dropped (queue full or packet too large)", (unsigned)d);
200 return q_.pop(buf, max);
201 }
202 bool connected() const override {
203 if (!has_peripheral_) return false;
204 try {
205 return const_cast<SimpleBLE::Peripheral&>(peripheral_).is_connected();
206 } catch (const std::exception& e) {
207 TINYBLEFS_LOGD("simpleble", "connection state unknown: %s", e.what());
208 return false;
209 }
210 }

214 if (a.size() != b.size()) return false;
215 for (size_t i = 0; i < a.size(); i++)
216 if (tolower((unsigned char)a[i]) != tolower((unsigned char)b[i]))
217 return false;
218 return true;
219 }

237 bool begin() {
238 try {
239 SimpleBLE::Adapter adapter;
240 if (!detail::selectAdapter(adapter)) return false;
241 using Cap = SimpleBLE::Local::CharacteristicCapability;
242 peripheral_ = adapter.create_local_peripheral();
243 peripheral_.add_advertised_service(kServiceUUID);
244 SimpleBLE::Local::Service svc = peripheral_.add_service(kServiceUUID);
245 rx_ = svc.add_characteristic(kRxUUID, {Cap::WRITE_REQUEST, Cap::WRITE_COMMAND});
246 tx_ = svc.add_characteristic(kTxUUID, {Cap::NOTIFY});
247
248 rx_.set_callback_on_write([this](SimpleBLE::ByteArray v) {
249 q_.push((const uint8_t*)v.data(), v.size());
250 });
251 tx_.set_callback_on_subscribed([this]() {
252 subscribed_ = true;
253 TINYBLEFS_LOGD("simpleble", "client subscribed to notifications");
254 });
255 tx_.set_callback_on_unsubscribed([this]() { subscribed_ = false; });
256 peripheral_.set_callback_on_client_connected([this](SimpleBLE::BluetoothAddress a) {
257 q_.clear();
258 connected_ = true;
259 TINYBLEFS_LOGI("simpleble", "client %s connected", a.c_str());
260 });
261 peripheral_.set_callback_on_client_disconnected([this](SimpleBLE::BluetoothAddress a) {
262 connected_ = false;
263 subscribed_ = false;
264 TINYBLEFS_LOGI("simpleble", "client %s disconnected", a.c_str());
265 });
266 peripheral_.start();
267 started_ = true;
268 TINYBLEFS_LOGI("simpleble", "advertising the TinyBLEFS service");
269 return true;
270 } catch (const std::exception& e) {
271 TINYBLEFS_LOGE("simpleble", "cannot start the GATT server: %s", e.what());
272 return false;
273 }
274 }
275
277 void end() {
278 if (!started_) return;
279 started_ = false;
280 try {
281 rx_.set_callback_on_write({});
282 peripheral_.stop();
283 } catch (const std::exception& e) {
284 TINYBLEFS_LOGD("simpleble", "stop: %s", e.what());
285 }
286 }
