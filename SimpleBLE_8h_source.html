36class LockedQueue {
37 public:
38 void push(const uint8_t* d, size_t n) {
39 std::lock_guard<std::mutex> g(m_);
40 if (q_.size() < 64) q_.emplace_back(d, d + n);
41 else dropped_++;
42 }
43 size_t pop(uint8_t* buf, size_t max) {
44 std::lock_guard<std::mutex> g(m_);
45 if (q_.empty()) return 0;
46 size_t n = std::min(max, q_.front().size());
47 memcpy(buf, q_.front().data(), n);
48 q_.pop_front();
49 return n;
50 }
51 void clear() {
52 std::lock_guard<std::mutex> g(m_);
53 q_.clear();
54 }
56 uint32_t takeDropped() {
57 std::lock_guard<std::mutex> g(m_);
58 uint32_t n = dropped_;
59 dropped_ = 0;
60 return n;
61 }
62
63 private:
64 std::mutex m_;
65 uint32_t dropped_ = 0;
66 std::deque<std::vector<uint8_t>> q_;
67};

78 static std::vector<SimpleBLE::Peripheral> scan(
79 int durationMs = 5000, const std::string& namePrefix = "") {
80 std::vector<SimpleBLE::Peripheral> found;
81 std::vector<SimpleBLE::Peripheral> results;
82 try {
83 auto adapters = SimpleBLE::Adapter::get_adapters();
84 if (adapters.empty()) {
85 TINYBLEFS_LOGE("simpleble", "no Bluetooth adapter found (is bluetoothd running?)");
86 return found;
87 }
88 SimpleBLE::Adapter adapter = adapters[0];
89 adapter.scan_for(durationMs);
90 results = adapter.scan_get_results();
91 } catch (const std::exception& e) {
92 TINYBLEFS_LOGE("simpleble", "scan failed: %s", e.what());
93 return found;
94 }
95 for (auto& p : results) {
96 bool tinyblefs = false;
97 for (auto& s : p.services())
98 if (equalsNoCase(s.uuid(), kServiceUUID)) tinyblefs = true;
99 if (!tinyblefs) continue;
100 if (!namePrefix.empty() && p.identifier().compare(0, namePrefix.size(), namePrefix) != 0)
101 continue;
102 found.push_back(p);
103 }
104 return found;
105 }

107 bool connect(SimpleBLE::Peripheral peripheral) {
108 disconnect();
109 try {
110 peripheral_ = peripheral;
111 peripheral_.connect();
112 has_peripheral_ = true;
113 TINYBLEFS_LOGI("simpleble", "connected, packets of %u bytes (ATT MTU %u)",
114 (unsigned)mtu(), (unsigned)peripheral_.mtu());
115 // after connect() BlueZ has resolved the services and negotiated the MTU
116 peripheral_.notify(kServiceUUID, kTxUUID, [this](SimpleBLE::ByteArray payload) {
117 q_.push((const uint8_t*)payload.data(), payload.size());

119 return true;
120 } catch (const std::exception& e) {
121 TINYBLEFS_LOGW("simpleble", "connect failed: %s", e.what());

123 return false;

139 // ---- Transport -------------------------------------------------------
140 size_t mtu() const override {
141 if (!has_peripheral_) return 20;
142 try {
143 uint16_t m = const_cast<SimpleBLE::Peripheral&>(peripheral_).mtu();
144 // ATT allows at most 517; BlueZ reports nonsense (65533) for some devices
145 if (m > 517) m = 517;
146 return m > 23 ? m - 3 : 20;
147 } catch (const std::exception& e) {
148 TINYBLEFS_LOGD("simpleble", "cannot read the MTU: %s", e.what());
149 return 20;
150 }
151 }
152 bool send(const uint8_t* data, size_t len) override {
153 if (!has_peripheral_) return false;
154 try {
155 peripheral_.write_command(kServiceUUID, kRxUUID,
156 SimpleBLE::ByteArray((const char*)data, len));
157 return true;
158 } catch (const std::exception& e) {
159 TINYBLEFS_LOGW("simpleble", "write of %u bytes failed: %s", (unsigned)len, e.what());
160 return false;
161 }
162 }
163 size_t receive(uint8_t* buf, size_t max) override {
164 uint32_t d = q_.takeDropped();
165 if (d) TINYBLEFS_LOGW("simpleble", "%u received packet(s) dropped (queue full or packet too large)", (unsigned)d);
166 return q_.pop(buf, max);
167 }
168 bool connected() const override {
169 if (!has_peripheral_) return false;
170 try {
171 return const_cast<SimpleBLE::Peripheral&>(peripheral_).is_connected();
172 } catch (const std::exception& e) {
173 TINYBLEFS_LOGD("simpleble", "connection state unknown: %s", e.what());
174 return false;
175 }
176 }

199 public:
200 ~SimpleBLEServerTransport() override { end(); }

203 bool begin() {
204 try {
205 auto adapters = SimpleBLE::Adapter::get_adapters();
206 if (adapters.empty()) {
207 TINYBLEFS_LOGE("simpleble", "no Bluetooth adapter found (is bluetoothd running?)");
208 return false;
209 }
210 using Cap = SimpleBLE::Local::CharacteristicCapability;
211 peripheral_ = adapters[0].create_local_peripheral();
212 peripheral_.add_advertised_service(kServiceUUID);
213 SimpleBLE::Local::Service svc = peripheral_.add_service(kServiceUUID);
214 rx_ = svc.add_characteristic(kRxUUID, {Cap::WRITE_REQUEST, Cap::WRITE_COMMAND});
215 tx_ = svc.add_characteristic(kTxUUID, {Cap::NOTIFY});
216
217 rx_.set_callback_on_write([this](SimpleBLE::ByteArray v) {
218 q_.push((const uint8_t*)v.data(), v.size());
219 });
220 tx_.set_callback_on_subscribed([this]() {
221 subscribed_ = true;
222 TINYBLEFS_LOGD("simpleble", "client subscribed to notifications");
223 });
224 tx_.set_callback_on_unsubscribed([this]() { subscribed_ = false; });
225 peripheral_.set_callback_on_client_connected([this](SimpleBLE::BluetoothAddress a) {
226 q_.clear();
227 connected_ = true;
228 TINYBLEFS_LOGI("simpleble", "client %s connected", a.c_str());
229 });
230 peripheral_.set_callback_on_client_disconnected([this](SimpleBLE::BluetoothAddress a) {
231 connected_ = false;
232 subscribed_ = false;
233 TINYBLEFS_LOGI("simpleble", "client %s disconnected", a.c_str());
234 });
235 peripheral_.start();
236 started_ = true;
237 TINYBLEFS_LOGI("simpleble", "advertising the TinyBLEFS service");
238 return true;
239 } catch (const std::exception& e) {
240 TINYBLEFS_LOGE("simpleble", "cannot start the GATT server: %s", e.what());
241 return false;
242 }
243 }
244
246 void end() {
247 if (!started_) return;
248 started_ = false;
249 try {
250 rx_.set_callback_on_write({});
251 peripheral_.stop();
252 } catch (const std::exception& e) {
253 TINYBLEFS_LOGD("simpleble", "stop: %s", e.what());
254 }
255 }

265 size_t mtu() const override { return packet_; }
266 bool send(const uint8_t* data, size_t len) override {
267 if (!connected_ || !subscribed_) {
268 TINYBLEFS_LOGW("simpleble", "cannot notify: no client connected or not subscribed");
269 return false;
270 }
271 try {
272 tx_.set_value(SimpleBLE::ByteArray((const char*)data, len)); // notifies
273 } catch (const std::exception& e) {
274 TINYBLEFS_LOGW("simpleble", "notification of %u bytes failed: %s", (unsigned)len, e.what());
275 return false;
276 }
277 if (pacing_ms_) std::this_thread::sleep_for(std::chrono::milliseconds(pacing_ms_));
278 return true;
279 }
280 size_t receive(uint8_t* buf, size_t max) override {
281 uint32_t d = q_.takeDropped();
282 if (d) TINYBLEFS_LOGW("simpleble", "%u received packet(s) dropped (queue full or packet too large)", (unsigned)d);
283 return q_.pop(buf, max);
284 }
285 bool connected() const override { return connected_; }
