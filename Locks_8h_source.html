2// Server side file locks: protect the file system (a file must not be opened
3// twice for writing: on FAT this corrupts the directory) and the data (two
4// appending clients would overwrite each other).
5//
6// Policy, per file:
7// - any number of readers
8// - one writer at a time (open with write, append, create or truncate)
9// - truncating (mode "w") needs a file that is not open at all
10// - remove / rename / rmdir need a path that is not open (nor any file below it)
11// Locks are released when the file is closed, the session ends or the client
12// disconnects.

25 bool acquire(const std::string& path, bool write, bool truncate) {
26 Entry& e = entries_[path];
27 bool ok = !write || (e.writers == 0 && (!truncate || e.readers == 0));
28 if (ok) {
29 if (write) e.writers++;
30 else e.readers++;
31 } else if (e.readers == 0 && e.writers == 0) {
32 entries_.erase(path);
33 }
34 return ok;
35 }
36
38 void release(const std::string& path, bool write) {
39 auto it = entries_.find(path);
40 if (it == entries_.end()) return;
41 uint16_t& n = write ? it->second.writers : it->second.readers;
42 if (n > 0) n--;
43 if (it->second.readers == 0 && it->second.writers == 0) entries_.erase(it);
44 }
45
47 bool isBusy(const std::string& path) const {
48 auto it = entries_.find(path);
49 if (it != entries_.end()) return true;
50 std::string prefix = path;
51 if (prefix.empty() || prefix.back() != '/') prefix += '/';
52 auto lb = entries_.lower_bound(prefix);
53 return lb != entries_.end() && lb->first.compare(0, prefix.size(), prefix) == 0;
54 }
