2// Server backend for any Arduino style file system class (ESP32 / RP2040
3// fs::FS: SD, SD_MMC, LittleFS, SPIFFS, FFat, ...).
4//
5// ArduinoFsBackend<fs::LittleFSFS> backend(LittleFS);
6// FileServer server(backend, transport, "sensor1");
7//
8// Requires the FS to accept string open modes ("r", "w", "r+") like the ESP32
9// and RP2040 cores do. Not usable with SdFat style O_* flags.
10//
11// On the ESP32 the directory listing and stat use the POSIX API of the virtual
12// file system (opendir / readdir / stat below FS::mountpoint()) instead of
13// opening a File object for every entry, which is much faster. The directory
14// is kept open between consecutive list() calls. Define
15// TINYBLEFS_ARDUINOFS_NO_POSIX to switch this off; TINYBLEFS_ARDUINOFS_POSIX enables
16// it on other systems whose FS has a mountpoint() (e.g. for tests).

48 FsFile* open(const char* path, uint8_t flags) override {
49 bool write = flags & (kWrite | kAppend | kCreate | kTrunc);
50 if (write) closeCursor();
51 bool exists = fs_.exists(path);
52 const char* mode = "r";
53 if (write) {
54 if (!exists && !(flags & (kCreate | kTrunc))) return nullptr;
55 mode = (!exists || (flags & kTrunc)) ? "w" : "r+";
56 } else if (!exists) {
57 return nullptr;
58 }
59 FileT f = fs_.open(path, mode);
60 if (!f || f.isDirectory()) {
61 TINYBLEFS_LOGE("arduinofs", "cannot open %s (mode %s)", path, mode);
62 return nullptr;
63 }
64 return new ArduinoFile(f);
65 }

68 bool stat(const char* path, FsEntry& e) override {
69#if defined(TINYBLEFS_ARDUINOFS_POSIX)
70 const char* mp = mountOf(fs_, 0);
71 if (mp && *mp) return statPosix(posixPath(mp, path), path, e);
72#endif
73 FileT f = fs_.open(path, "r");
74 if (!f) return false;
75 e.name = baseName(f.name());
76 e.isDir = f.isDirectory();
77 e.size = e.isDir ? 0 : (uint32_t)f.size();
78 f.close();
79 return true;
80 }

82 bool list(const char* path, uint32_t index, FsEntry& e) override {
83#if defined(TINYBLEFS_ARDUINOFS_POSIX)
84 const char* mp = mountOf(fs_, 0);
85 if (mp && *mp) return listPosix(posixPath(mp, path), index, e);
86#endif
87 FileT dir = fs_.open(path, "r");
88 if (!dir || !dir.isDirectory()) return false;
89 for (uint32_t i = 0;; i++) {
90 FileT item = dir.openNextFile();
91 if (!item) break;
92 if (i == index) {
93 e.name = baseName(item.name());
94 e.isDir = item.isDirectory();
95 e.size = e.isDir ? 0 : (uint32_t)item.size();
96 item.close();
97 dir.close();
98 return true;
99 }
100 item.close();
101 }
102 dir.close();
103 return false;
104 }

184 static bool statPosix(const std::string& full, const char* path, FsEntry& e) {
185 if (path[0] == 0 || (path[0] == '/' && path[1] == 0)) { // the root directory
186 e.name.clear();
187 e.isDir = true;
188 e.size = 0;
189 return true;
190 }
191 struct stat st;
192 if (::stat(full.c_str(), &st) != 0) return false;
193 e.name = baseName(path);
194 e.isDir = S_ISDIR(st.st_mode);
195 e.size = e.isDir ? 0 : (uint32_t)st.st_size;
196 return true;
197 }

200 struct dirent* nextEntry() {
201 struct dirent* de;
202 while ((de = readdir(cursor_)) != nullptr) {
203 if (strcmp(de->d_name, ".") != 0 && strcmp(de->d_name, "..") != 0) break;
204 }
205 return de;
206 }

208 bool listPosix(const std::string& dir, uint32_t index, FsEntry& e) {
209 // The server asks for entry 0, 1, 2, ...: continue where we stopped
210 if (!cursor_ || cursor_next_ != index || cursor_path_ != dir) {
211 closeCursor();
212 cursor_ = opendir(dir.c_str());
213 if (!cursor_) {
214 TINYBLEFS_LOGD("arduinofs", "list %s: cannot open directory", dir.c_str());
215 return false;
216 }
217 cursor_path_ = dir;
218 cursor_next_ = 0;
219 while (cursor_next_ < index) { // not sequential: skip the first entries
220 if (!nextEntry()) {
221 closeCursor();
222 return false;
223 }
224 cursor_next_++;
225 }
226 }
227 struct dirent* de = nextEntry();
228 if (!de) { // end of the directory
229 closeCursor();
230 return false;
231 }
232 cursor_next_++;
233 e.name = de->d_name;
234 std::string full = dir;
235 if (full.empty() || full.back() != '/') full += '/';
236 full += de->d_name;
237 struct stat st;
238 bool have = ::stat(full.c_str(), &st) == 0; // the size is not in dirent
239 e.isDir = de->d_type == DT_DIR || (de->d_type == DT_UNKNOWN && have && S_ISDIR(st.st_mode));
240 e.size = (e.isDir || !have) ? 0 : (uint32_t)st.st_size;
241 return true;
242 }
