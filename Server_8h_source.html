148 void route(const uint8_t* p, size_t n) {
149 if (p[0] == 0) {
150 control(p, n);
151 return;
152 }
153 auto it = sessions_.find(p[0]);
154 if (it == sessions_.end()) {
155 TINYBLEFS_LOGD("session", "packet for unknown session %u", (unsigned)p[0]);
156 sendControl(kSessionExpired, p[0]);
157 return;
158 }
159 it->second->lastSeen = nowMs();
160 if (n > 1) it->second->link.push(p + 1, n - 1);
161 }
162
163 void control(const uint8_t* p, size_t n) {
164 if (n < 2) return;
165 if (p[1] == kSessionHello && n >= 6) {
166 uint32_t nonce = get32(&p[2]);
167 for (auto& kv : sessions_) { // repeated hello: same client, same session
168 if (kv.second->nonce == nonce) {
169 kv.second->lastSeen = nowMs();
170 sendWelcome(nonce, kv.first);
171 return;
172 }
173 }
174 if (sessions_.size() >= max_sessions_) {
175 TINYBLEFS_LOGW("session", "no free session (%u sessions): hello refused", (unsigned)max_sessions_);
176 std::vector<uint8_t> r{0, kSessionFull};
177 put32(r, nonce);
178 raw_.send(r.data(), r.size());
179 return;
180 }
181 uint8_t sid = nextId();
182 auto s = std::make_unique<Session>(fs_, raw_, sid, nonce, name_, root_, max_handles_, &locks_);
183 s->server.setReadOnly(read_only_);
184 s->server.setMaxListBytes(max_list_bytes_);
185 sessions_[sid] = std::move(s);
186 TINYBLEFS_LOGI("session", "session %u opened (%u of %u)", (unsigned)sid,
187 (unsigned)sessions_.size(), (unsigned)max_sessions_);
188 sendWelcome(nonce, sid);
189 } else if (p[1] == kSessionKeepAlive && n >= 3) {
190 auto it = sessions_.find(p[2]);
191 if (it != sessions_.end()) it->second->lastSeen = nowMs();
192 else sendControl(kSessionExpired, p[2]);
193 }
194 }

198 uint8_t nextId() {
199 for (int i = 0; i < 255; i++) {
200 last_id_ = (uint8_t)(last_id_ % 255 + 1);
201 if (!sessions_.count(last_id_)) return last_id_;
202 }
203 return 0; // not reached: size < max_sessions_ <= 255
204 }
