24class PacketQueue {
25 public:
26 void push(const uint8_t* d, size_t n) {
27 std::lock_guard<std::mutex> g(m_);
28 if (q_.size() < 64) q_.emplace_back(d, d + n);
29 else dropped_++;
30 }
32 uint32_t takeDropped() {
33 std::lock_guard<std::mutex> g(m_);
34 uint32_t n = dropped_;
35 dropped_ = 0;
36 return n;
37 }
38 size_t pop(uint8_t* buf, size_t max) {
39 std::lock_guard<std::mutex> g(m_);
40 if (q_.empty()) return 0;
41 size_t n = q_.front().size() < max ? q_.front().size() : max;
42 memcpy(buf, q_.front().data(), n);
43 q_.pop_front();
44 return n;
45 }
46 void clear() {
47 std::lock_guard<std::mutex> g(m_);
48 q_.clear();
49 }
50
51 private:
52 std::mutex m_;
53 uint32_t dropped_ = 0;
54 std::deque<std::vector<uint8_t>> q_;
55};
56} // namespace detail

60 public NimBLEServerCallbacks,
61 public NimBLECharacteristicCallbacks {
62 public:
68 void setConnectionInterval(uint16_t minUnits, uint16_t maxUnits) {

70 conn_max_ = maxUnits;

77 bool begin(const char* deviceName, bool secure = false) {
78 if (!NimBLEDevice::isInitialized()) NimBLEDevice::init(deviceName);
79 NimBLEDevice::setMTU(517);
80 if (secure) NimBLEDevice::setSecurityAuth(true, false, true);
81
82 server_ = NimBLEDevice::createServer();
83 server_->setCallbacks(this, false);
84 NimBLEService* svc = server_->createService(kServiceUUID);
85 if (!svc) {
86 TINYBLEFS_LOGE("nimble", "cannot create the TinyBLEFS service");
87 return false;
88 }
89 uint32_t enc = secure ? (NIMBLE_PROPERTY::READ_ENC | NIMBLE_PROPERTY::WRITE_ENC) : 0;
90 rx_ = svc->createCharacteristic(
91 kRxUUID, NIMBLE_PROPERTY::WRITE | NIMBLE_PROPERTY::WRITE_NR | enc);
92 tx_ = svc->createCharacteristic(
93 kTxUUID, NIMBLE_PROPERTY::NOTIFY | (secure ? NIMBLE_PROPERTY::READ_ENC : 0));
94 rx_->setCallbacks(this);
95 svc->start();
96
97 NimBLEAdvertising* adv = NimBLEDevice::getAdvertising();
98 adv->addServiceUUID(kServiceUUID);
99 // The 128 bit service UUID fills most of the 31 byte advertisement, and
100 // NimBLE-Arduino 2.x does not add the name by itself: put it into the scan
101 // response so that scanners (and the name prefix of scan()) can see it.
102 NimBLEAdvertisementData scanResponse;
103 scanResponse.setName(deviceName);
104 adv->setScanResponseData(scanResponse);
105 adv->enableScanResponse(true);
106 if (!adv->start()) {
107 TINYBLEFS_LOGE("nimble", "cannot start advertising");
108 return false;
109 }
110 TINYBLEFS_LOGI("nimble", "advertising as '%s'", deviceName);
111 return true;
112 }
113
114 // ---- Transport -------------------------------------------------------
115 size_t mtu() const override {
116 if (conn_ == BLE_HS_CONN_HANDLE_NONE) return 20;
117 uint16_t m = server_->getPeerMTU(conn_);
118 size_t n = m > 23 ? m - 3 : 20;
119 return n > kNimBleMaxPacket ? kNimBleMaxPacket : n;
120 }
121 bool send(const uint8_t* data, size_t len) override {
122 for (int retry = 0; retry < 200; retry++) { // mbufs may be exhausted
123 if (conn_ == BLE_HS_CONN_HANDLE_NONE) return false;
124 if (tx_->notify(data, len, conn_)) return true;
125 delay(2);
126 }
127 TINYBLEFS_LOGW("nimble", "notification of %u bytes failed (no free buffers or "
128 "client not subscribed)", (unsigned)len);
129 return false;
130 }
131 size_t receive(uint8_t* buf, size_t max) override {
132 reportDropped();
133 return rx_q_.pop(buf, max);
134 }
135 bool connected() const override { return conn_ != BLE_HS_CONN_HANDLE_NONE; }
136
137 private:
138 void onConnect(NimBLEServer*, NimBLEConnInfo& info) override {
139 if (conn_ != BLE_HS_CONN_HANDLE_NONE) {
140 // one client at a time: a second one would mix up the first one's data
141 TINYBLEFS_LOGW("nimble", "second client refused: the server handles one client at a time");
142 server_->disconnect(info.getConnHandle());
143 return;
144 }
145 conn_ = info.getConnHandle();
146 TINYBLEFS_LOGI("nimble", "client connected");
147 rx_q_.clear();
148 // fast connection interval for throughput (7.5 .. 15 ms)
149 server_->updateConnParams(conn_, conn_min_, conn_max_, 0, 200);
150 // data length extension: 251 bytes per link layer packet instead of 27
151 server_->setDataLen(conn_, 251);
152 }
153 void onDisconnect(NimBLEServer*, NimBLEConnInfo& info, int reason) override {
154 if (info.getConnHandle() != conn_) { // a refused client
155 NimBLEDevice::startAdvertising();
156 return;
157 }
158 TINYBLEFS_LOGI("nimble", "client disconnected (reason %d)", reason);
159 conn_ = BLE_HS_CONN_HANDLE_NONE;
160 NimBLEDevice::startAdvertising();
161 }
162 void onWrite(NimBLECharacteristic* c, NimBLEConnInfo&) override {
163 NimBLEAttValue v = c->getValue();
164 rx_q_.push(v.data(), v.size());
165 }
166
167 void reportDropped() {
168 uint32_t n = rx_q_.takeDropped();
169 if (n) TINYBLEFS_LOGW("nimble", "%u received packet(s) dropped (queue full or packet too large)", (unsigned)n);
170 }
171
172 uint16_t conn_min_ = 6, conn_max_ = 6;
173 NimBLEServer* server_ = nullptr;
174 NimBLECharacteristic *rx_ = nullptr, *tx_ = nullptr;
175 volatile uint16_t conn_ = BLE_HS_CONN_HANDLE_NONE;
176 detail::PacketQueue rx_q_;
177};

189 static std::vector<NimBLEAddress> scan(uint32_t durationMs = 5000,
190 const char* namePrefix = "") {
191 size_t prefixLen = namePrefix ? strlen(namePrefix) : 0;
192 if (!NimBLEDevice::isInitialized()) NimBLEDevice::init("");
193 std::vector<NimBLEAddress> found;
194 NimBLEScan* sc = NimBLEDevice::getScan();
195 sc->setActiveScan(true);
196 NimBLEScanResults res = sc->getResults(durationMs);
197 NimBLEUUID uuid(kServiceUUID);
198 for (int i = 0; i < res.getCount(); i++) {
199 const NimBLEAdvertisedDevice* d = res.getDevice(i);
200 if (!d->isAdvertisingService(uuid)) continue;
201 if (prefixLen && d->getName().compare(0, prefixLen, namePrefix) != 0)
202 continue;
203 found.push_back(d->getAddress());
204 }
205 return found;
206 }
207
208 bool connect(const NimBLEAddress& address) {
209 disconnect();
210 if (!NimBLEDevice::isInitialized()) NimBLEDevice::init("");
211 NimBLEDevice::setMTU(517);
212 client_ = NimBLEDevice::createClient();
213 if (!client_) return fail("cannot create a BLE client (too many connections?)");
214 if (!client_->connect(address)) return fail("cannot connect to the device");
215 client_->setDataLen(251); // data length extension
216 client_->updateConnParams(6, 12, 0, 200); // 7.5 - 15 ms connection interval
217 NimBLERemoteService* svc = client_->getService(kServiceUUID);
218 if (!svc) return fail("device has no TinyBLEFS service");
219 rx_ = svc->getCharacteristic(kRxUUID);
220 NimBLERemoteCharacteristic* tx = svc->getCharacteristic(kTxUUID);
221 if (!rx_ || !tx) return fail("TinyBLEFS characteristics not found");
222 detail::PacketQueue* q = &rx_q_;
223 if (!tx->subscribe(true, [q](NimBLERemoteCharacteristic*, uint8_t* d,
224 size_t n, bool) { q->push(d, n); }))
225 return fail("cannot subscribe to notifications");
226 return true;
227 }
228
229 void disconnect() {
230 if (client_) {
231 client_->disconnect();
232 NimBLEDevice::deleteClient(client_);
233 client_ = nullptr;
234 }
235 rx_ = nullptr;
236 rx_q_.clear();
237 }
238
239 // ---- Transport -------------------------------------------------------
240 size_t mtu() const override {
241 uint16_t m = client_ ? client_->getMTU() : 23;
242 size_t n = m > 23 ? m - 3 : 20;
243 return n > kNimBleMaxPacket ? kNimBleMaxPacket : n;
244 }
245 bool send(const uint8_t* data, size_t len) override {
246 if (rx_ && rx_->writeValue(data, len, false)) return true;
247 TINYBLEFS_LOGW("nimble", "write of %u bytes failed", (unsigned)len);
248 return false;
249 }
250 size_t receive(uint8_t* buf, size_t max) override {
251 uint32_t n = rx_q_.takeDropped();
252 if (n) TINYBLEFS_LOGW("nimble", "%u received packet(s) dropped (queue full or packet too large)", (unsigned)n);
253 return rx_q_.pop(buf, max);
254 }
255 bool connected() const override { return client_ && client_->isConnected(); }
256
257 private:
258 bool fail(const char* why) {
259 TINYBLEFS_LOGW("nimble", "connect failed: %s", why);
260 disconnect();
261 return false;
262 }
263 NimBLEClient* client_ = nullptr;
264 NimBLERemoteCharacteristic* rx_ = nullptr;
265 detail::PacketQueue rx_q_;
266};
