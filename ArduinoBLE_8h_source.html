2// Transport adapters for the ArduinoBLE library (UNO R4 WiFi, Nano 33 BLE,
3// Nano 33 IoT, MKR WiFi 1010, Nicla, ...).
4//
5// Server: tinyblefs::ArduinoBLEServerTransport (one client at a time)
6// Client: tinyblefs::ArduinoBLEClientTransport (one object per remote server)
7//
8// ArduinoBLE is polled: Server::poll() and the Client call Transport::poll(),
9// which runs BLE.poll(). Don't block in loop() for long.
10//
11// ArduinoBLE can't tell the negotiated ATT MTU, so the packet size defaults to
12// the safe 20 bytes. If both sides negotiate a larger MTU (e.g. 247) call
13// setPacketSize(244) for much better throughput.

109 static void onWritten(BLEDevice, BLECharacteristic c) {
110 if (!self()) return;
111 static uint8_t tmp[TINYBLEFS_RX_PACKET]; // static: this runs deep inside BLE.poll() on a small stack
112 int n = c.readValue(tmp, sizeof(tmp));
113 if (n > 0) self()->q_.push(tmp, n); // overflow is reported by receive()
114 }
115 static void onConnected(BLEDevice) {
116 ArduinoBLEServerTransport* t = self();
117 if (!t) return;
118 if (t->connections_ == 0) t->q_.clear();
119 t->connections_++;
120 TINYBLEFS_LOGI("arduinoble", "client connected (%u of %u)", (unsigned)t->connections_,
121 (unsigned)t->max_);
122 if (t->connections_ < t->max_) BLE.advertise(); // room for more clients
123 }
124 static void onDisconnected(BLEDevice) {
125 ArduinoBLEServerTransport* t = self();
126 if (!t) return;
127 if (t->connections_ > 0) t->connections_--;
128 TINYBLEFS_LOGI("arduinoble", "client disconnected (%u left)", (unsigned)t->connections_);
129 BLE.advertise(); // ArduinoBLE stops advertising while connected
130 }

153 static std::vector<BLEDevice> scan(uint32_t durationMs = 5000,
154 const char* namePrefix = "") {
155 std::vector<BLEDevice> found;
156 if (!detail::beginArduinoBLE()) {
157 TINYBLEFS_LOGE("arduinoble", "BLE.begin() failed: no BLE hardware?");
158 return found;
159 }
160 BLE.scanForUuid(kServiceUUID);
161 uint32_t start = millis();
162 while (millis() - start < durationMs) {
163 BLEDevice d = BLE.available();
164 if (!d) continue;
165 if (namePrefix && *namePrefix && !d.localName().startsWith(namePrefix))
166 continue;
167 bool known = false; // devices are reported repeatedly
168 for (auto& f : found) known = known || f.address() == d.address();
169 if (!known) found.push_back(d);
170 }
171 BLE.stopScan();
172 return found;
173 }
174
176 bool connect(BLEDevice device) {
177 disconnect();
178 if (!detail::beginArduinoBLE()) return fail("BLE.begin() failed");
179 device_ = device;
180 if (!device_.connect()) return fail("cannot connect to the device");
181 if (!device_.discoverAttributes()) return fail("service discovery failed");
182 rx_ = device_.characteristic(kRxUUID);
183 tx_ = device_.characteristic(kTxUUID);
184 if (!rx_ || !tx_) return fail("device has no TinyBLEFS characteristics");
185 if (!tx_.canSubscribe()) return fail("notifications are not supported");
186 tx_.setEventHandler(BLEUpdated, onUpdated);
187 if (!tx_.subscribe()) return fail("cannot subscribe to notifications");
188 instances().push_back(this);
189 registered_ = true;
190 return true;
191 }
192
194 void disconnect() {
195 if (registered_) {
196 auto& v = instances();
197 for (size_t i = 0; i < v.size(); i++)
198 if (v[i] == this) { v.erase(v.begin() + i); break; }
199 registered_ = false;
200 }
201 if (device_ && device_.connected()) device_.disconnect();
202 q_.clear();
203 }
204
206 void setPacketSize(size_t n) {
207 packet_ = n > TINYBLEFS_RX_PACKET ? TINYBLEFS_RX_PACKET : (n < 20 ? 20 : n);
208 }

228 return v;
229 }
231 static void onUpdated(BLEDevice device, BLECharacteristic c) {
232 static uint8_t tmp[TINYBLEFS_RX_PACKET]; // static: small stack, see onWritten
233 int n = c.readValue(tmp, sizeof(tmp));
234 if (n <= 0) return;
235 for (auto* t : instances())
236 if (t->device_.address() == device.address()) t->q_.push(tmp, n);
237 }
238 bool fail(const char* why) {
239 TINYBLEFS_LOGW("arduinoble", "connect failed: %s", why);
240 disconnect();
241 return false;
242 }
